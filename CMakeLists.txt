cmake_minimum_required(VERSION 3.20)
project(Chronos VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)

include(CTest)
include(FetchContent)
option(CHRONOS_ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(CHRONOS_ENABLE_BENCHMARKS "Fetch Google Benchmark for future benchmarks" OFF)

add_library(chronos_core src/orderbook/order_book.cpp src/matching/matching_engine.cpp)
target_compile_features(chronos_core PUBLIC cxx_std_20)
target_include_directories(chronos_core PUBLIC $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/include>)

function(chronos_configure_target target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion)
  endif()
  if(CHRONOS_ENABLE_SANITIZERS)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" OR WIN32)
      message(FATAL_ERROR "Sanitizers require GCC or Clang on a supported non-Windows platform")
    endif()
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endfunction()
chronos_configure_target(chronos_core)

add_executable(chronos_cli apps/exchange_cli/main.cpp)
target_link_libraries(chronos_cli PRIVATE chronos_core)
target_compile_definitions(chronos_cli PRIVATE CHRONOS_VERSION="${PROJECT_VERSION}")
chronos_configure_target(chronos_cli)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(Threads REQUIRED)
  add_library(chronos_gateway_lib src/gateway/protocol.cpp src/gateway/tcp.cpp)
  target_compile_features(chronos_gateway_lib PUBLIC cxx_std_20)
  target_include_directories(chronos_gateway_lib PUBLIC $<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/include>)
  target_link_libraries(chronos_gateway_lib PUBLIC Threads::Threads)
  chronos_configure_target(chronos_gateway_lib)
  add_executable(chronos_gateway apps/gateway/main.cpp)
  target_link_libraries(chronos_gateway PRIVATE chronos_gateway_lib)
  chronos_configure_target(chronos_gateway)
  add_executable(chronos_gateway_client apps/gateway_client/main.cpp)
  target_link_libraries(chronos_gateway_client PRIVATE chronos_gateway_lib)
  chronos_configure_target(chronos_gateway_client)
endif()

if(BUILD_TESTING)
  FetchContent_Declare(googletest
    GIT_REPOSITORY https://github.com/google/googletest.git
    GIT_TAG f8d7d77c06936315286eb55f8de22cd23c188571)
  set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(googletest)
  add_subdirectory(tests)
endif()

if(CHRONOS_ENABLE_BENCHMARKS)
  set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(googlebenchmark
    GIT_REPOSITORY https://github.com/google/benchmark.git
    GIT_TAG 344117638c8ff7e239044fd0fa7085839fc03021)
  FetchContent_MakeAvailable(googlebenchmark)
  add_subdirectory(benchmarks)
endif()
